package edu.odu.cs.cs350;

/**
 * 
 * Calculate the EPS for a semester based on enrollment data 
 */
	public class CalculateEPS {
		// TODO Auto-generated method stub
		
		
		public void CalculateEPS() {
			
			
		}
		/**
		 * Calculates the change in enrollment between two snapshots.
		 *
		 * @param previousOffering enrollment in the earlier snapshot
		 * @param currentOffering enrollment in the later snapshot
		 * @return the change in student enrollment
		 * @throws IllegalArgumentException if enrollment is negative
		 */
		public int calculateEnrollmentChanges( Offering previousOffering, Offering currentOffering) {
			int previousEnrollment = previousOffering.getTotalEnrollment();
		    int currentEnrollment = currentOffering.getTotalEnrollment();
		    
		    //If the total Enrollment of the previous Offering or the current Offering is negative
			if (previousEnrollment < 0 || currentEnrollment < 0) {
			        throw new IllegalArgumentException( "Enrollment in one or more Offering is negative.");
			    }
			    
			    //returns the delta (-) if it lowered (+) if it grew
			    return currentEnrollment - previousEnrollment;
			}
		
		/**
		 * Calculates if the enrollment change is positive between two offerings.
		 *
		 * @param previousOffering enrollment in the earlier snapshot
		 * @param currentOffering enrollment in the later snapshot
		 * @return true if enrollment has increased, false if it decreased
		 */
		public boolean hasEnrollmentIncreased( Offering previousOffering, Offering currentOffering) {
				return calculateEnrollmentChanges(previousOffering,currentOffering) > 0;
		}
		
		/**
		 * Calculates if the enrollment change is  negative between two offerings.
		 *
		 * @param previousOffering enrollment in the earlier snapshot
		 * @param currentOffering enrollment in the later snapshot
		 * @return true if enrollment has decreased, false if it increased
		 */
		public boolean hasEnrollmentDecreased( Offering previousOffering, Offering currentOffering) {
			return calculateEnrollmentChanges(previousOffering,currentOffering) < 0;
		}
		
		public boolean isEnrollmentSame( Offering previousOffering, Offering currentOffering) {
			return calculateEnrollmentChanges(previousOffering,currentOffering) == 0;
		}
	}
