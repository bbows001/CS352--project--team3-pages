package edu.odu.cs.cs350;

import java.io.IOException;
import java.io.OutputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

import org.apache.poi.ss.usermodel.Row;
import org.apache.poi.ss.usermodel.Sheet;
import org.apache.poi.ss.usermodel.Workbook;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

/** Class to generate a detailed Excel report. */

public class OutputDetailedReport {
    /**
     * Creates a detailed Excel report generator.
     */
    public OutputDetailedReport() {
    }

    /**
     * Create Excel Workbook for enrollment history.
     * 
     * @return the Excel workbook
     */

    public Workbook createWorkbook() {
        Workbook workbook = new XSSFWorkbook();
        Sheet sheet = workbook.createSheet("Current Semester History");
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("Course");
        return workbook;

    }

    /**
     * Add snapshot dates as headers for the columns on the worksheet.
     * 
     * @param workbook the Excel workbook
     * @param dates    the snapshot dates
     */
    public void addSnapshotHeaders(Workbook workbook, List<LocalDate> dates) {
        Sheet sheet = workbook.getSheet("Current Semester History");
        Row header = sheet.getRow(0);
        int column = 1;
        for (LocalDate date : dates) {
            header.createCell(column).setCellValue(date.toString());
            column++;
        }
    }

    /**
     * Add course and enrollment integers to excel report.
     * 
     * @param workbook    Excel workbook
     * @param courseName  the course Name
     * @param enrollment1 and enrollment total for first snapshot
     * @param enrollment2 enrollment total for second snapshot
     */

    public void addCourseEnrollment(Workbook workbook, String courseName, int enrollment1,
            int enrollment2) {

        Sheet sheet = workbook.getSheet("Current Semester History");
        int nextRow = sheet.getLastRowNum() + 1;
        Row row = sheet.createRow(nextRow);
        row.createCell(0).setCellValue(courseName);
        row.createCell(1).setCellValue(enrollment1);
        row.createCell(2).setCellValue(enrollment2);

    }

    /**
     * Adds course and course enrollment history to Excel wksht.
     * 
     * @param workbook    the Excel workbook
     * @param courseName  the name of the course
     * @param enrollments the enrollment totals for the snapshot
     */

    public void addCourseEnrollment(Workbook workbook, String courseName, List<Integer> enrollments) {

        Sheet sheet = workbook.getSheet("Current Semester History");
        int nextRow = sheet.getLastRowNum() + 1;
        Row row = sheet.createRow(nextRow);
        row.createCell(0).setCellValue(courseName);
        int column = 1;
        for (int enrollment : enrollments) {
            row.createCell(column).setCellValue(enrollment);
            column++;
        }
    }

    /**
     * Save Excel workbook to a file.
     * 
     * @param workbook the workbook to be saved
     * @param filePath the location of the Excel doc
     * @throws IOException if the file cant be written
     */
    public void saveWorkbook(Workbook workbook, Path filePath) throws IOException {
        try (OutputStream output = Files.newOutputStream(filePath)) {
            workbook.write(output);

        }
    }

    /**
     * Adds enrollment history for all courses in all snapshots.
     * 
     * @param workbook  the Excel workbook
     * @param snapshots the enrollment snapshots
     */
    public void addSectionEnrollmentHistory(Workbook workbook, List<List<Section>> snapshots) {
        List<String> previouslyAddedCourses = new ArrayList<>();

        // Look through each snapshot for courses
        for (List<Section> snapshot : snapshots) {
            for (Section section : snapshot) {

                String courseName = section.getCourseName();

                // Skip courses already added to the report
                if (previouslyAddedCourses.contains(courseName)) {
                    continue;
                }

                previouslyAddedCourses.add(courseName);

                List<Integer> enrollments = new ArrayList<>();

                // Calc enrollment for this course at each date
                for (List<Section> currentSnapshot : snapshots) {
                    int enrollment = 0;

                    for (Section current : currentSnapshot) {
                        if (current.getCourseName().equals(courseName)) {
                            enrollment += current.getCurrentEnrollment();
                        }
                    }
                    enrollments.add(enrollment);
                }

                // Add the course and course enrollment history to Excel
                addCourseEnrollment(workbook, courseName, enrollments);
            }
        }
    }
}
