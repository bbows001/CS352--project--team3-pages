package edu.odu.cs.cs350;
import java.io.IOException;
import java.time.LocalDate;
import java.time.temporal.ChronoUnit;

/**
 * Represents an enrollment window with start and end dates.
 */
public class EnrollmentWindow {
    private LocalDate startDate;
    private LocalDate endDate;

    /**
     * Default constructor for the EnrollmentWindow class.
     * Initializes startDate and endDate to null.
     */
    public EnrollmentWindow() {
        this.startDate = null;
        this.endDate = null;
    }

    /**
     * Constructor for the EnrollmentWindow class.
     * Initializes startDate and endDate to the provided values.
     *
     * @param startDate the start date of the enrollment window
     * @param endDate   the end date of the enrollment window
     */
    public EnrollmentWindow(LocalDate startDate, LocalDate endDate) {
        this.startDate = startDate;
        this.endDate = endDate;
        validateDates(); // Validate the dates upon creation
    }

    /**
     * Sets the start date of the enrollment window.
     *
     * @param startDate the start date to set
     */
    public void setStartDate(LocalDate startDate) {
        this.startDate = startDate;
    }

    /**
     * Gets the start date of the enrollment window.
     *
     * @return the start date
     */
    public LocalDate getStartDate() {
        return startDate;
    }

    /**
     * Sets the end date of the enrollment window.
     *
     * @param endDate the end date to set
     */
    public void setEndDate(LocalDate endDate) {
        this.endDate = endDate;
    }  

    /**
     * Gets the end date of the enrollment window.
     *
     * @return the end date
     */
    public LocalDate getEndDate() {
        return endDate;
    }

    /**
     * Calculates the duration of the enrollment window in days.
     *
     * @return the duration in days
     */
    public int getDuration() {
        if (startDate != null && endDate != null) {
            return (int) java.time.temporal.ChronoUnit.DAYS.between(startDate, endDate);
        }
        return 0;
    }

    /**
     * Throws an IllegalArgumentException if the end date is before the start date.
     * 
     * @throws IllegalArgumentException if the end date is before the start date
     */
    public void validateDates() {
        if (startDate != null && endDate != null && endDate.isBefore(startDate)) {
            throw new IllegalArgumentException("End date cannot be before start date.");
        }  
    }
    /**Calculates the elapsed percentage of the enrollment window
     * @param currentDate and the start/end dates of the enrollment window
     * @return the percentage elapsed of the enrollment period
     * @throws IllegalArgumentException if the dates are invalid
     * 
     */
    public double getElapsedEnrollmentPercent(LocalDate currentDate){
        if (startDate ==null || endDate == null){
            throw new IllegalArgumentException("Invalid start or end date.");
        }
        if (currentDate == null || currentDate.isBefore(startDate) || currentDate.isAfter(endDate)){
            throw new IllegalArgumentException("Current date is not within the enrollment window.");
        }
        double totalDays = ChronoUnit.DAYS.between(startDate, endDate);
        double elapsedDays = ChronoUnit.DAYS.between(startDate, currentDate);
        double elapsedPercent = (elapsedDays/totalDays) *100;
        return elapsedPercent;
        }
    }
