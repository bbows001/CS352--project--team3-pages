package edu.odu.cs.cs350;

import java.time.LocalDate;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;
import java.util.logging.Logger;

/**
 * Class to print/output summary of enrollment data.
 * 
 */
public class OutputSummaryReport {
        /**
         * Creates a summary enrollment report generator.
         */
        public OutputSummaryReport() {

        }

        /** Logger used for error and warning messages in this class. */
        private static final Logger LOGGER = Logger.getLogger(OutputSummaryReport.class.getName());

        /**
         * Displays the enrollment period elapsed.
         * 
         * @param enrollmentWindow the enrollment period being analyzed
         * @param snapshotDate the date of the enrollment snapshot
         * 
         */

        public void displayEnrollmentProgress(EnrollmentWindow enrollmentWindow, LocalDate snapshotDate) {

                System.out.println("Enrollment Window Elapsed Percentage: " +
                                enrollmentWindow.getElapsedEnrollmentPercent(snapshotDate) + "%");
        }

        /**
         * Builds one row of the summary report for a course. The row starts with "*"
         * if the projected enrollment is more than the capacity, or a blank space
         * otherwise.
         *
         * @param courseName    the name of the course
         * @param enrollment    the course's current total enrollment
         * @param projected     the course's projected enrollment
         * @param totalCapacity the course's total capacity
         * @return the formatted report row
         */
        public String formatCourseRow(String courseName, int enrollment, int projected, int totalCapacity) {
                String prefix = " ";
                if (projected > totalCapacity) {
                        prefix = "*";
                }
                return prefix + courseName + "\t" + enrollment + "\t" + projected + "\t" + totalCapacity;
        }

        /**
         * Displays the enrollment summary for one course.
         *
         * @param courseName    the name of the course
         * @param enrollment    the current enrollment
         * @param projected     the projected enrollment
         * @param totalCapacity the total course capacity
         */
        public void displayCourseSummary(String courseName, int enrollment,
                        int projected, int totalCapacity) {

                // Print the column Headings
                System.out.println("Course\tCurrent\tProjected\tCapacity");

                // Print the formatted course information
                System.out.println(
                                formatCourseRow(
                                                courseName,
                                                enrollment,
                                                projected,
                                                totalCapacity));
        }

        /**
         * Displays the enrollment summary for a single section.
         *
         * For Sprint 1, projected enrollment is temporarily
         * equal to current enrollment.
         *
         * @param section the section to display
         */
        public void displaySectionSummary(Section section) {

                String courseName = section.getCourseName();
                int enrollment = section.getCurrentEnrollment();
                int totalCapacity = section.getCapacity();

                // Temporary projection for Sprint 1 TODO
                int projected = enrollment;
                // Provide values to the display method
                displayCourseSummary(
                                courseName,
                                enrollment,
                                projected,
                                totalCapacity);
        }

        /**
         * Displays a summary of courses by combining sections belonging
         * to the same course.
         *
         * @param sections the sections to summarize
         */
        public void displayMultipleCourses(List<Section> sections) {

                // Group sections by course name
                Map<String, Offering> courses = new TreeMap<>();

                for (Section section : sections) {
                        // Include lectures and standalone sections.
                        // Exclude linked recitations and labs.
                        if (section.hasLink() && !section.isLecture()) {
                                continue;
                        }

                        String courseName = section.getCourseName();

                        // Create an Offering if this course is not in the map
                        if (!courses.containsKey(courseName)) {
                                courses.put(courseName, new Offering());
                        }

                        // Add the eligible section
                        courses.get(courseName).addSection(section);
                }

                // Print the headings once
                System.out.println("Course\tCurrent\tProjected\tCapacity");

                // Display one summary row per course
                for (Map.Entry<String, Offering> entry : courses.entrySet()) {

                        String courseName = entry.getKey();
                        Offering offering = entry.getValue();

                        int enrollment = offering.getTotalEnrollment();
                        int totalCapacity = offering.getTotalCapacity();

                        // Temporary Sprint 1 projection
                        int projected = enrollment;

                        System.out.println(
                                        formatCourseRow(
                                                        courseName,
                                                        enrollment,
                                                        projected,
                                                        totalCapacity));
                }
        }

        /**
         * Displays the enrollment summary from the last snapshot
         * in a list of snapshots. Logs a warning if there are no snapshots.
         *
         * @param snapshots enrollment snapshots in
         *                  chronological order
         */
        public void displayLatestSnapshot(
                        List<List<Section>> snapshots) {

                // Handle the case where no snapshots were loaded
                if (snapshots == null || snapshots.isEmpty()) {
                        LOGGER.warning("No enrollment snapshots available.");
                        return;
                }

                // Retrieve the last snapshot
                List<Section> latestSnapshot = snapshots.get(snapshots.size() - 1);

                // Display the retrieved snapshot course summary
                displayMultipleCourses(latestSnapshot);
        }

}
