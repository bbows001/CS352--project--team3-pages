package edu.odu.cs.cs350;

/**
 * Represents a section of a course.
 */
public class Section {
	
	String courseName;
	String CRN;
	String Subject;

	/**
	 * Default constructor for the Section class.
	 * Initializes courseName, CRN, and Subject to empty strings.
	 */
	public Section() {
		this.courseName = "";
		this.CRN = "";
		this.Subject = "";
	}

	/**
	 * Constructor for the Section class.
	 * Initializes courseName to the provided value and CRN and Subject to empty strings.
	 * @param courseName the name of the course
	 */
	public Section(String courseName) {
		this.courseName = courseName;
		this.CRN = "";
		this.Subject = "";
	}

	/**
	 * Sets the name of the course.
	 * @param courseName the name of the course
	 */
	public void setCourseName(String courseName) {
		this.courseName = courseName;
	}

	/**
	 * Gets the name of the course.
	 * @return the name of the course
	 */
	public String getCourseName() {
		return courseName;
	}

	/**
	 * Sets the subject of the section.
	 * @param Subject the subject of the section
	 */
	public void setSubject(String Subject) {
		this.Subject = Subject;
	}

	/**
	 * Gets the subject of the section.
	 * @return the subject of the section
	 */
	public String getSubject() {
		return Subject;
	}

	/**
	 * Sets the CRN of the section.
	 * @param CRN the CRN of the section
	 */
	public void setCRN(String CRN) {
		this.CRN = CRN;
	}

	/**
	 * Gets the CRN of the section.
	 * @return the CRN of the section
	 */
	public String getCRN() {
		return CRN;
	}
	

}
