package edu.odu.cs.cs350;

/**
 * Represents a section of a course.
 */
public class Section {
	
	private String courseName;
	private String CRN;
	private String subject;
	private String campus;
	private int capacity;
	private int currentEnrollment;
	private String crossList;
	private int overallCapacity;
	private int overallEnrollment;
	private String link;

	/**
	 * Default constructor for the Section class.
	 * Initializes courseName, CRN, and Subject to empty strings.
	 */
	public Section() {
		this("", "", "", "", 0, 0, "");
	}

	/**
	 * Constructor for the Section class.
	 * Initializes courseName to the provided value and sets CRN 
	 * and Subject to empty strings.
	 * 
	 * @param courseName the name of the course
	 */
	public Section(String courseName) {
		this(courseName, "", "", "", 0, 0, "");
	}

	/**
	 * Constructor for the Section class.
	 * Initializes courseName, CRN, and Subject to the provided values.
	 * 
	 * @param courseName the name of the course
	 * @param CRN the CRN of the section
	 * @param subject the subject of the section
	 * @param campus the campus of the section
	 * @param capacity the capacity of the section
	 * @param currentEnrollment the number of students currently enrolled in the section
	 * @param crossList the cross-list section #
	 */
	public Section(String courseName, String CRN, String subject, String campus, int capacity, int currentEnrollment, String crossList) {
		this.courseName = courseName;
		this.CRN = CRN;
		this.subject = subject;
		this.campus = campus;
		this.capacity = capacity;
		this.currentEnrollment = currentEnrollment;
		this.crossList = crossList;
	}

	/**
	 * Sets the name of the course.
	 * 
	 * @param courseName the name of the course
	 */
	public void setCourseName(String courseName) {
		this.courseName = courseName;
	}

	/**
	 * Gets the name of the course.
	 * @return the name of the course
	 */
	public String getCourseName() {
		return courseName;
	}

	/**
	 * Sets the subject of the section.
	 * @param subject the subject of the section
	 */
	public void setSubject(String subject) {
		this.subject = subject;
	}

	/**
	 * Gets the subject of the section.
	 * @return the subject of the section
	 */
	public String getSubject() {
		return subject;
	}

	/**
	 * Sets the CRN of the section.
	 * @param CRN the CRN of the section
	 */
	public void setCRN(String CRN) {
		this.CRN = CRN;
	}

	/**
	 * Gets the CRN of the section.
	 * @return the CRN of the section
	 */
	public String getCRN() {
		return CRN;
	}
	
	/**
	 * Sets the campus of the section.
	 * @param campus the campus of the section
	 */
	public void setCampus(String campus) {
		this.campus = campus;	
	}

	/**
	 * Gets the campus of the section.
	 * @return the campus of the section
	 */
	public String getCampus() {
		return campus;
	}

	/**
	 * Sets the capacity of the section.
	 * @param capacity the capacity of the section
	 */
	public void setCapacity(int capacity) {
		this.capacity = capacity;
	}

	/**
	 * Gets the capacity of the section.
	 * @return the capacity of the section
	 */
	public int getCapacity() {
		return capacity;
	}

	/**
	 * Sets the overall capacity of the offering associated with this section.
	 * 
	 * @param overallCapacity the overall offering capacity
	 */
	public void setOverallCapacity(int overallCapacity) {
		this.overallCapacity = overallCapacity;
	}

	/**
 	*Gets the overall capacity of the offering associated with this section. 
	*
 	* @return the overall offering capacity
 	*/
	public int getOverallCapacity() {
		return overallCapacity;
	}


	/**
	 * Sets the number of students currently enrolled in the section.
	 * @param currentEnrollment the number of students currently enrolled
	 */
	public void setCurrentEnrollment(int currentEnrollment) {
		this.currentEnrollment = currentEnrollment;
	}

	/**
	 * Gets the number of students currently enrolled in the section.
	 * @return the number of students currently enrolled
	 */
	public int getCurrentEnrollment() {
		return currentEnrollment;
	}

	/**
	 * Sets the overall enrollment of the offering associated with this section.
	 * 
	 * @param overallEnrollment the overall offering enrollment
	 */
	public void setOverallEnrollment(int overallEnrollment) {
		this.overallEnrollment = overallEnrollment;
	}

	/**
	 * Gets the overall enrollment of the offering associated with this section.
	 * 
	 * @return the overall offering enrollment
	 */
	public int getOverallEnrollment() {
		return overallEnrollment;
	}

	/**
	 * Sets the cross-list section number.
	 * @param crossList the cross-list section number
	 */
	public void setCrossList(String crossList) {
		this.crossList = crossList;
	}

	/**
	 * Gets the cross-list section number.
	 * @return the cross-list section number
	 */
	public String getCrossList() {
		return crossList;
	}

	/**
	 * Sets the LINK identifier for this section.
	 * 
	 * @param link the section's LINK identifier
	 */
	public void setLink(String link) {
		this.link = link;
	}

	/**
	 * Gets the LINK identifier for this section.
	 * 
	 * @return the section's LINK identifier
	 */
	public String getLink() {
		return link;
	}

	/**
 	 * Determines whether this section is a lecture based 
	 * on its LINK identifier.
 	 *
 	 * @return true if the LINK identifier ends in 1, false otherwise
	 */
	public boolean isLecture() {
    	return link != null && link.endsWith("1");
	}

	/**
 	 * Determines whether this section is a recitation based 
	 * on its LINK identifier.
 	 *
 	 * @return true if the LINK identifier ends in 2, false otherwise
	 */
	public boolean isRecitation() {
    	return link != null && link.endsWith("2");
	}

	/**
 	 * Determines whether this section is a lab based 
	 * on its LINK identifier.
 	 *
 	 * @return true if the LINK identifier ends in 3, false otherwise
	 */
	public boolean isLab() {
    	return link != null && link.endsWith("3");
	}

	/**
	 * Determines whether this section has a LINK identifier.
	 * 
	 * @return true if the LINK identifier is not null or blank,
	 * false otherwise.
	 */
	public boolean hasLink() {
		return link != null && !link.isBlank();
	}
}
