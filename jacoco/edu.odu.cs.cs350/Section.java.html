package edu.odu.cs.cs350;

/**
 * Represents a section of a course.
 */
public class Section {
	
	private String courseName;
	private String CRN;
	private String subject;
	private String campus;
	private int capacity;
	private int numCurrentlyEnrolled;
	private int crossList;

	/**
	 * Default constructor for the Section class.
	 * Initializes courseName, CRN, and Subject to empty strings.
	 */
	public Section() {
		this("", "", "", "", 0, 0, 0);
	}

	/**
	 * Constructor for the Section class.
	 * Initializes courseName to the provided value and sets CRN 
	 * and Subject to empty strings.
	 * 
	 * @param courseName the name of the course
	 */
	public Section(String courseName) {
		this(courseName, "", "", "", 0, 0, 0);
	}

	/**
	 * Constructor for the Section class.
	 * Initializes courseName, CRN, and Subject to the provided values.
	 * 
	 * @param courseName the name of the course
	 * @param CRN the CRN of the section
	 * @param subject the subject of the section
	 * @param campus the campus of the section
	 * @param capacity the capacity of the section
	 * @param numCurrentlyEnrolled the number of students currently enrolled in the section
	 * @param crossList the cross-list section #
	 */
	public Section(String courseName, String CRN, String subject, String campus, int capacity, int numCurrentlyEnrolled, int crossList) {
		this.courseName = courseName;
		this.CRN = CRN;
		this.subject = subject;
		this.campus = campus;
		this.capacity = capacity;
		this.numCurrentlyEnrolled = numCurrentlyEnrolled;
		this.crossList = crossList;
	}

	/**
	 * Sets the name of the course.
	 * 
	 * @param courseName the name of the course
	 */
	public void setCourseName(String courseName) {
		this.courseName = courseName;
	}

	/**
	 * Gets the name of the course.
	 * @return the name of the course
	 */
	public String getCourseName() {
		return courseName;
	}

	/**
	 * Sets the subject of the section.
	 * @param subject the subject of the section
	 */
	public void setSubject(String subject) {
		this.subject = subject;
	}

	/**
	 * Gets the subject of the section.
	 * @return the subject of the section
	 */
	public String getSubject() {
		return subject;
	}

	/**
	 * Sets the CRN of the section.
	 * @param CRN the CRN of the section
	 */
	public void setCRN(String CRN) {
		this.CRN = CRN;
	}

	/**
	 * Gets the CRN of the section.
	 * @return the CRN of the section
	 */
	public String getCRN() {
		return CRN;
	}
	
	/**
	 * Sets the campus of the section.
	 * @param campus the campus of the section
	 */
	public void setCampus(String campus) {
		this.campus = campus;	
	}

	/**
	 * Gets the campus of the section.
	 * @return the campus of the section
	 */
	public String getCampus() {
		return campus;
	}

	/**
	 * Sets the capacity of the section.
	 * @param capacity the capacity of the section
	 */
	public void setCapacity(int capacity) {
		this.capacity = capacity;
	}

	/**
	 * Gets the capacity of the section.
	 * @return the capacity of the section
	 */
	public int getCapacity() {
		return capacity;
	}

	/**
	 * Sets the number of students currently enrolled in the section.
	 * @param numCurrentlyEnrolled the number of students currently enrolled
	 */
	public void setNumCurrentlyEnrolled(int numCurrentlyEnrolled) {
		this.numCurrentlyEnrolled = numCurrentlyEnrolled;
	}

	/**
	 * Gets the number of students currently enrolled in the section.
	 * @return the number of students currently enrolled
	 */
	public int getNumCurrentlyEnrolled() {
		return numCurrentlyEnrolled;
	}

	/**
	 * Sets the cross-list section number.
	 * @param crossList the cross-list section number
	 */
	public void setCrossList(int crossList) {
		this.crossList = crossList;
	}

	/**
	 * Gets the cross-list section number.
	 * @return the cross-list section number
	 */
	public int getCrossList() {
		return crossList;
	}

}
