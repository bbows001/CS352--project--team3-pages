package edu.odu.cs.cs350;

import java.io.IOException;
import java.nio.file.Path;
import java.time.LocalDate;
import java.time.format.DateTimeParseException;
import java.util.ArrayList;
import java.util.List;
import java.util.logging.Logger;

import org.apache.poi.ss.usermodel.Workbook;

/**
 * Main class for Enrollment Projection System.
 * Program start, reads enrollment data from semester folder
 * Determines enrollment period, displays enrollment progress
 */
public class EnrollmentProjectionSystem {
    /**
     * Creates the enrollment projection system.
     */
    public EnrollmentProjectionSystem() {

    }

    /** Logger used for error messages from the main program. */
    private static final Logger LOGGER = Logger.getLogger(EnrollmentProjectionSystem.class.getName());

    /**
     * Starts the Enrollment Projection System and reads and validates the
     * enrollment data.
     *
     * @param semesterDataLocation from the command line: the semester folder,
     *                             optionally
     *                             followed by an enrollment cutoff date
     *                             (YYYY-MM-DD)
     * @throws IOException if there is no semester folder or an error reading the
     *                     folder/files
     */
    public static void main(String[] semesterDataLocation) throws IOException {
        // Check if user provides a semester folder
        // If not, log an error and stop the program

        if (semesterDataLocation.length == 0) {
            LOGGER.severe("Please provide a semester folder.");
            return;
        }
        // Get folder location given by the user
        Path semesterFolder = Path.of(semesterDataLocation[0]);

        // Optional enrollment cutoff date given after the folder
        // Snapshots dated after this date will be ignored
        LocalDate cutoffDate = null;
        if (semesterDataLocation.length > 1) {
            try {
                cutoffDate = LocalDate.parse(semesterDataLocation[1]);
            } catch (DateTimeParseException e) {
                LOGGER.severe("Invalid cutoff date (use YYYY-MM-DD): " + semesterDataLocation[1]);
                throw e;
            }
        }

        // Create a reader object to access the semester files
        ReadSemesterEnrollmentData reader = new ReadSemesterEnrollmentData();

        // Read enrollment snapshot files from the folder
        // Each one has a list of course section and enrollment info for specific date
        // Section = One course section and its enrollment information
        // List<Section> = All the sections recorded in one snapshot
        // List<List<Section>> = A collection of all those snapshots
        List<List<Section>> snapshots;
        try {
            snapshots = reader.readSemester(semesterFolder, cutoffDate);
        } catch (IOException e) {
            LOGGER.severe(e.getMessage());
            throw e;
        }

        // Display number of enrollment snapshots loaded
        System.out.println("Snapshots loaded: " + snapshots.size());

        // Read the semester start and end date from the date file
        List<LocalDate> dates = reader.readDates(semesterFolder);

        // Create an enrollment window object to track enrollment period start/end date
        EnrollmentWindow window = new EnrollmentWindow();

        // Set beginning and end of enrollment dates by retrieving dates from list
        window.setStartDate(dates.get(0));
        window.setEndDate(dates.get(1));

        // Create object report to display info about enrollment progress
        OutputSummaryReport report = new OutputSummaryReport();

        // Temp placeholder date to use as snapshot, update later
        LocalDate snapshotDate = LocalDate.of(2026, 5, 19);

        // Display how far along the enrollment period is based on snapshot
        // date and enrollment window
        report.displayEnrollmentProgress(window, snapshotDate);

        // Last change for Sprint 1
        // Display course enrollment from the latest loaded CSV
        report.displayLatestSnapshot(snapshots);

        // Create detailed Excel enrollment report
        OutputDetailedReport detailedReport = new OutputDetailedReport();
        try (Workbook workbook = detailedReport.createWorkbook()) {
            List<LocalDate> snapshotDates = new ArrayList<>();
            boolean foundAfterEndDate = false;
            for (Path file : reader.locateEnrollmentSnapshots(semesterFolder)) {

                String fileName = file.getFileName().toString();
                String dateText = fileName.replace(".csv", "");
                LocalDate date = LocalDate.parse(dateText);

                // Skip snapshots after cutoff date
                if (cutoffDate != null && date.isAfter(cutoffDate)) {
                    continue;
                }
                // Skip snapshots before enrollment begins
                if (date.isBefore(window.getStartDate())) {
                    continue;
                }
                // Include only the first snapshot after enrollment ends
                if (date.isAfter(window.getEndDate())) {
                    if (foundAfterEndDate) {
                        continue;
                    }
                    foundAfterEndDate = true;
                }
                snapshotDates.add(date);

            }
            // Add dates and enrollment data to excel workbook, then save
            detailedReport.addSnapshotHeaders(workbook, snapshotDates);
            detailedReport.addSectionEnrollmentHistory(workbook, snapshots);
            detailedReport.saveWorkbook(workbook, Path.of("EnrollmentReport.xlsx"));

        }
    }
}
