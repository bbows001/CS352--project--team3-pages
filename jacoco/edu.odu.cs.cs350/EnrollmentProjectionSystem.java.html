package edu.odu.cs.cs350;

import java.io.IOException;
import java.nio.file.Path;
import java.time.LocalDate;
import java.util.List;

/**
 * Main class for Enrollment Projection System
 */
public class EnrollmentProjectionSystem {

    /**
     * Starts the Enrollment Projection System and reads and validates the
     * enrollment data
     * 
     * @param args from the command line
     * @throws IOException if there is no semester folder or an error reading the
     *                     info
     */
    public static void main(String[] args) throws IOException {

        if (args.length == 0) {
            System.out.println("Please provide a semester folder.");
            return;
        }
        Path semesterFolder = Path.of(args[0]);
        ReadSemesterEnrollmentData reader = new ReadSemesterEnrollmentData();
        List<List<Section>> snapshots = reader.readSemester(semesterFolder);
        System.out.println("Snapshots loaded: " + snapshots.size());
        List<LocalDate> dates = reader.readDates(semesterFolder);

        EnrollmentWindow window = new EnrollmentWindow();
        window.setStartDate(dates.get(0));
        window.setEndDate(dates.get(1));
        OutputSummaryReport report = new OutputSummaryReport();
        LocalDate snapshotDate = LocalDate.of(2026, 5, 19); // replace with actual date later
        report.displayEnrollmentProgress(window, snapshotDate);
    }

}
