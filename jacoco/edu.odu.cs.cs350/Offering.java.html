package edu.odu.cs.cs350;

import java.util.ArrayList;
import java.util.List;

/**
 * Represents an offering of a course.
 */
public class Offering {

	private String instructor;
	private List<Section> sections;
	private int term;

	/**
	 * Default constructor for the Offering class.
	 * Initializes instructor to an empty string.
	 * Default term is Fall (10)
	 */
	public Offering() {
		this.instructor = "";
		this.term = 10; //Default term is Fall (10). Other options are Spring (20) or Summer (30).
		this.sections = new ArrayList<>();
	}

	/**
	 * Constructor for the Offering class.
	 * Initializes instructor to the provided value.
	 * 
	 * @param instructor the name of the instructor
	 */
	public Offering(String instructor) {
		this.instructor = instructor;
		this.term = 10;
		this.sections = new ArrayList<>();
	}
	
	/**
	 * Constructor for the Offering class.
	 * Initializes instructor to the provided value.
	 * 
	 * @param instructor the name of the instructor
	 * @param term the name of the term of the offering
	 */
	public Offering(String instructor, int term) {
		this.instructor = instructor;
		setTerm(term);
		this.sections = new ArrayList<>();
	}
	
	/**
	 * Constructor for the Offering class.
	 * Initializes instructor to the provided value.
	 * 
	 * @param instructor, the name of the instructor
	 * @param term, the name of the term of the offering
	 * @param sections, the list of sections in the offering
	 */
	public Offering(String instructor, int term, List<Section> sections) {
		this.instructor = instructor;
		this.term = term;
		this.sections = new ArrayList<>(sections);
	}

	// Setters and Getters
	/**
	 * Sets the instructor for the offering.
	 * 
	 * @param instructor the instructor to set
	 */
	public void setInstructor(String instructor) {
		this.instructor = instructor;
	}
	
	/**
	 * Gets the instructor for the offering.
	 * 
	 * @return the instructor
	 */
	public String getInstructor() {
		return instructor;
	}
	
	/**
	 * Sets the term for the offering and validates its inputs.
	 * 
	 * @param term, the name of the term of the offering
	 */
	public void setTerm(int term) {
		// TODO Auto-generated method stub
		if (term != 10 && term !=20 && term != 30) {
			throw new IllegalArgumentException("Invalid term. Term must be either Fall (10), Spring (20) or Summer (30).");
		}
		else {
			this.term = term;
		}
	}
	
	/**
	 * Gets the term for the offering.
	 * 
	 * @return the term
	 */
	public int getTerm() {
	    return term;
	}
	
	/**
	 * Adds a new section to the the offering.
	 * 
	 * @param section, an individual section inside of the lists of sections in the offering
	 */
	public void addSection(Section section) {
		// TODO Auto-generated method stub
		sections.add(section);
	}
	
	/**
	 * Gets the full list of sections from the offering.
	 * 
	 * @return all the sections
	 */
	public List<Section> getAllSections(){
		return sections;
	}
	
	/**
	 * Removes an existing section to the the offering.
	 * 
	 * @param section, an individual section inside of the lists of sections in the offering
	 */
	public void removeSection(Section section) {
		// TODO Auto-generated method stub
		sections.remove(section);
	}

	/**
	 * Gets the total enrollment of all sections from the offering.
	 * 
	 * @return the total enrolled students
	 */
	public int getTotalEnrollment() {
		// TODO Auto-generated method stub
		int totalEnrolled = 0;  
		for (Section s : sections) {
		       totalEnrolled += s.getNumCurrentlyEnrolled();
		    }
		return totalEnrolled;
	}

	/**
	 * Gets the total capacity of all sections from the offering.
	 * 
	 * @return the total capacity of all sections in an offering
	 */
	public int getTotalCapacity() {
		// TODO Auto-generated method stub
		int totalCapacity = 0;  
		for (Section s : sections) {
		       totalCapacity += s.getCapacity();
		    }
		return totalCapacity;
	}

	/**
	 * Gets a specific section from the offering based on the CRN of said section.
	 * 
	 * @param CRN, the CRN of the section
	 * @return null if the offering is empty or if the search does not find the CRN or the section whose CRN matches the one in the parameter.
	 */
	public Section getSectionsByCRN(String CRN) {
		// TODO Auto-generated method stub
		if (sections.isEmpty()) {
			System.out.println("This offering has no sections.");
			return null;
		}else {
			for(Section s: sections) {
				if (s.getCRN().equals(CRN)) {
					return s;
				}
			}
		}
		System.out.println("The section with CRN: '" +CRN+ "' could not be found.");
		return null;
	}

	/**
	 * Gets a the number of sections that are at capacity or above capacity.
	 * 
	 * @return the number of section that were identified as being at capacity of above capacity. 
	 */
	public int sectionsAtCapacity() {
		// TODO Auto-generated method stub
		List<Section> maxedSections = new ArrayList<>(); 
		for(Section s:sections) {
			if(s.getNumCurrentlyEnrolled() >= s.getCapacity()) {
				maxedSections.add(s);
			}
		}
		return maxedSections.size();
	}
}
