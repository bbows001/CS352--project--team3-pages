package edu.odu.cs.cs350;

/**
 * Represents an offering of a course.
 */
public class Offering {

	private String instructor;

	/**
	 * Default constructor for the Offering class.
	 * Initializes instructor to an empty string.
	 */
	public Offering() {
		this.instructor = "";
	}

	/**
	 * Constructor for the Offering class.
	 * Initializes instructor to the provided value.
	 * @param string the name of the instructor
	 */
	public Offering(String string) {
		this.instructor = string;
	}

	// Setters and Getters
	/**
	 * Sets the instructor for the offering.
	 * @param instructor the instructor to set
	 */
	public void setInstructor(String instructor) {
		this.instructor = instructor;
	}

	/**
	 * Gets the instructor for the offering.
	 * @return the instructor
	 */
	public String getInstructor() {
		return instructor;
	}

}
