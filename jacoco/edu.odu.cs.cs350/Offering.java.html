package edu.odu.cs.cs350;

import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.SortedMap;
import java.util.TreeMap;
import java.util.logging.Logger;

/**
 * Represents an offering of a course.
 */
public class Offering {

        /** Logger used for error and warning messages in this class. */
        private static final Logger LOGGER = Logger.getLogger(Offering.class.getName());

        private String instructor;
        private List<Section> sections;
        private int term;

        /**
         * Default constructor for the Offering class.
         * Initializes instructor to an empty string.
         * Default term is Fall (10)
         */
        public Offering() {
                this.instructor = "";
                this.term = 10; //Default term is Fall (10). Other options are Spring (20) or Summer (30).
                this.sections = new ArrayList<>();
        }

        /**
         * Constructor for the Offering class.
         * Initializes instructor to the provided value.
         * 
         * @param instructor the name of the instructor
         */
        public Offering(String instructor) {
                this.instructor = instructor;
                this.term = 10;
                this.sections = new ArrayList<>();
        }

        /**
         * Constructor for the Offering class.
         * Initializes instructor to the provided value.
         * 
         * @param instructor the name of the instructor
         * @param term the name of the term of the offering
         */
        public Offering(String instructor, int term) {
                this.instructor = instructor;
                setTerm(term);
                this.sections = new ArrayList<>();
        }

        /**
         * Constructor for the Offering class.
         * Initializes instructor to the provided value.
         * 
         * @param instructor, the name of the instructor
         * @param term, the name of the term of the offering
         * @param sections, the list of sections in the offering
         */
        public Offering(String instructor, int term, List<Section> sections) {
                this.instructor = instructor;
                this.term = term;
                this.sections = new ArrayList<>(sections);
        }

        // Setters and Getters
        /**
         * Sets the instructor for the offering.
         * 
         * @param instructor the instructor to set
         */
        public void setInstructor(String instructor) {
                this.instructor = instructor;
        }

        /**
         * Gets the instructor for the offering.
         * 
         * @return the instructor
         */
        public String getInstructor() {
                return instructor;
        }

        /**
         * Sets the term for the offering and validates its inputs.
         * 
         * @param term, the name of the term of the offering
         */
        public void setTerm(int term) {
                if (term != 10 && term !=20 && term != 30) {
                        String message = "Invalid term. Term must be either Fall (10), Spring (20) or Summer (30).";
                        LOGGER.severe(message);
                        throw new IllegalArgumentException(message);
                }
                else {
                        this.term = term;
                }
        }

        /**
         * Gets the term for the offering.
         * 
         * @return the term
         */
        public int getTerm() {
            return term;
        }

        /**
         * Adds a new section to the the offering.
         * 
         * @param section, an individual section inside of the lists of sections in the offering
         */
        public void addSection(Section section) {
                sections.add(section);
        }

        /**
         * Gets the full list of sections from the offering.
         * 
         * @return all the sections
         */
        public List<Section> getAllSections(){
                return sections;
        }

        /**
         * Removes an existing section to the the offering.
         * 
         * @param section, an individual section inside of the lists of sections in the offering
         */
        public void removeSection(Section section) {
                sections.remove(section);
        }

        /**
         * Gets the total enrollment of all sections from the offering.
         * 
         * @return the total enrolled students
         */
        public int getTotalEnrollment() {
                int totalEnrolled = 0;  
                for (Section s : sections) {
                       totalEnrolled += s.getCurrentEnrollment();
                    }
                return totalEnrolled;
        }

        /**
         * Calculates the total capacity of an offering without 
         * double-counting cross-listed sections.
         *
         * @return the total capacity of the offering
         */
        public int getTotalCapacity() {
                int totalCapacity = 0;

                Set<String> countedGroups = new HashSet<>();

                for (Section section : sections) {
                        String crossList = section.getCrossList();
                        // Check for if we have a crosslist
                        if (crossList != null && !crossList.isBlank()) {
                                // Count the shared capacity only once per group.
                                if (countedGroups.add(crossList)) {
                                int overallCapacity = section.getOverallCapacity();

                                if (overallCapacity > 0) {
                                        totalCapacity += overallCapacity;
                                } else {
                                        totalCapacity += section.getCapacity();
                                }
                                }
                        } else {
                                // Independent sections use their individual capacity.
                                totalCapacity += section.getCapacity();
                        }
                }

                return totalCapacity;
        }

        /**
         * Gets a specific section from the offering based on the CRN of said section.
         * Logs a warning if the offering is empty or the CRN is not found.
         * 
         * @param CRN, the CRN of the section
         * @return null if the offering is empty or if the search does not find the CRN or the section whose CRN matches the one in the parameter.
         */
        public Section getSectionsByCRN(String CRN) {
                if (sections.isEmpty()) {
                        LOGGER.warning("This offering has no sections.");
                        return null;
                }else {
                        for(Section s: sections) {
                                if (s.getCRN().equals(CRN)) {
                                        return s;
                                }
                        }
                }
                LOGGER.warning("The section with CRN: '" + CRN + "' could not be found.");
                return null;
        }

        /**
         * Gets a the number of sections that are at capacity or above capacity.
         * 
         * @return the number of section that were identified as being at capacity of above capacity. 
         */
        public int sectionsAtCapacity() {
                List<Section> maxedSections = new ArrayList<>(); 
                for(Section s:sections) {
                        if(s.getCurrentEnrollment() >= s.getCapacity()) {
                                maxedSections.add(s);
                        }
                }
                return maxedSections.size();
        }

        /**
         * Groups a snapshot's lecture sections into one Offering per course, sorted by course name.
         *
         * @param snapshot, the snapshot that gets organize by courses
         * @return courses, the course names mapped to the Offering holding their sections
         */
        public static SortedMap<String, Offering> groupOfferingByCourse(List<Section> snapshot) {

                SortedMap<String, Offering> courses = new TreeMap<>();

                for (Section s : snapshot) {
                        // Bypasses Labs and Recitations, only categorizing the Lectures and unlinked sections
                        if (s.isLab() || s.isRecitation()) {
                                continue;
                        }

                        String courseName = s.getCourseName();

                        // Looks up the course; if it isn't in the map yet, creates and stores a new Offering for it
                        Offering offering = courses.get(courseName);
                        if (offering == null) {
                                offering = new Offering();
                                courses.put(courseName, offering);
                        }

                        offering.addSection(s);
                }
                return courses;
        }
}
