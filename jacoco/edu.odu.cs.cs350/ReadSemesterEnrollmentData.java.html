package edu.odu.cs.cs350;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.DirectoryStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

import org.apache.commons.csv.CSVFormat;
import org.apache.commons.csv.CSVParser;
import org.apache.commons.csv.CSVRecord; 


//Read semster enrollment info
public class ReadSemesterEnrollmentData {
	public List<LocalDate> readDates(Path semesterFolder) throws IOException{
		Path filePath = semesterFolder.resolve("dates.txt");
		if(!Files.exists(filePath)){
			throw new IOException("Missing dates.txt in " + semesterFolder.getFileName());
		}
		List<String> lines = Files.readAllLines(filePath);
		 if(lines.size() !=2){
			throw new IOException("Invalid number of dates in " + semesterFolder.getFileName());
		 }
		List<LocalDate> dates = new java.util.ArrayList<>();
		for (String line : lines){
			dates.add(LocalDate.parse(line));
		}
		return dates;
	}

    public List<Section> readEnrollmentSnapshot(Path csvTestFile) throws IOException {
		List<Section> sections = new ArrayList<>();
		try(CSVParser parser = CSVParser.parse(csvTestFile, StandardCharsets.UTF_8, 
			CSVFormat.DEFAULT.withFirstRecordAsHeader().withAllowMissingColumnNames())){
		for (CSVRecord row : parser){
			
			String crn = row.get("CRN");
            String subject = row.get("SUBJ");
            String courseNumber = row.get("CRSE");
            String campus = row.get("CAMPUS");

            int capacity = Integer.parseInt(row.get("XLST CAP"));
            int enrollment = Integer.parseInt(row.get("ENR"));
			Section section = new Section( 
				subject + courseNumber, crn, subject, campus, capacity, enrollment, 0);
			sections.add(section);
		}}
			return sections;
		}
		public List<Path> locateEnrollmentSnapshots(Path semesterFolder) throws IOException {
			List<Path> files = new ArrayList<>();
			try(DirectoryStream<Path> stream = Files.newDirectoryStream(semesterFolder, "*.csv")){
				
		for (Path file : stream){
			files.add(file);

		}		}
		files.sort(null);
		return files;
	}}
