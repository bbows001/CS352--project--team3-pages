package edu.odu.cs.cs350;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.DirectoryStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

import org.apache.commons.csv.CSVFormat;
import org.apache.commons.csv.CSVParser;
import org.apache.commons.csv.CSVRecord;

/**
 * Reads/validates semester enrollment data from CSV files and date information.
 */
public class ReadSemesterEnrollmentData {
/** Read/validate the semester date from data text file
 * @param semesterFolder Path to the semester folder
 * @return List of LocalDate objects representing the semester start and end dates
 * @throws IOException if the dates.txt file is missing or has an invalid number of dates
*/
	public List<LocalDate> readDates(Path semesterFolder) throws IOException {
		Path filePath = semesterFolder.resolve("dates.txt");
		if (!Files.exists(filePath)) {
			throw new IOException("Missing dates.txt in " + semesterFolder.getFileName());
		}
		List<String> lines = Files.readAllLines(filePath);
		if (lines.size() != 2) {
			throw new IOException("Invalid number of dates in " + semesterFolder.getFileName());
		}
		List<LocalDate> dates = new ArrayList<>();
		for (String line : lines) {
			dates.add(LocalDate.parse(line));
		}
		return dates;
	}

	/** Read a CSV enrollment snapshot and return a list of section objects
	 * @param csvTestFile path to the enrollment snapshot CSV file
	 * @return list of section objects
	 * @throws IOException if the file has bad data or missing data
	 */
	public List<Section> readEnrollmentSnapshot(Path csvTestFile) throws IOException {
		List<Section> sections = new ArrayList<>();
		try (CSVParser parser = CSVParser.parse(csvTestFile, StandardCharsets.UTF_8,
				CSVFormat.DEFAULT.withFirstRecordAsHeader().withAllowMissingColumnNames())) {
			for (CSVRecord row : parser) {

				String crn = row.get("CRN");
				String subject = row.get("SUBJ");
				String courseNumber = row.get("CRSE");
				String campus = row.get("CAMPUS");

				int capacity = Integer.parseInt(row.get("XLST CAP"));
				int enrollment = Integer.parseInt(row.get("ENR"));
				Section section = new Section(
						subject + courseNumber, crn, subject, campus, capacity, enrollment, 0);
				sections.add(section);
			}
		}
		return sections;
	}

	/** Find/sort CSV snapshot files in a semester folder
	 * @param semesterFolder path to the semester directory
	 * @return Array list of snapshot files
	 * @throws IOException if the file folder can not be read
	 * */

	public List<Path> locateEnrollmentSnapshots(Path semesterFolder) throws IOException {
		List<Path> files = new ArrayList<>();
		try (DirectoryStream<Path> stream = Files.newDirectoryStream(semesterFolder, "*.csv")) {

			for (Path file : stream) {
				files.add(file);
			}
		}
		files.sort(null);
		return files;
	}

	/** Load snapshot for specified date range 
	 * @param semesterFolder containing the snapshots and date files
	 * @return array list of snapshots 
	 * @throws IOException if the folder is blank or missing files
	*/

	public List<List<Section>> readSemester(Path semesterFolder) throws IOException {
		List<List<Section>> snapshots = new ArrayList<>();
		List<Path> files = locateEnrollmentSnapshots(semesterFolder);
		List<LocalDate> dates = readDates(semesterFolder);
		LocalDate startDate = dates.get(0);
		LocalDate endDate = dates.get(1);
		boolean foundAfterEndDate = false;
		for (Path file : files) {
			String fileName = file.getFileName().toString();
			String dateString = fileName.replace(".csv", "");
			LocalDate snapshotDate = LocalDate.parse(dateString);
			if (snapshotDate.isBefore(startDate)) {
				continue;
			}
			if (snapshotDate.isAfter(endDate)) {

				if (foundAfterEndDate) {
					continue;
				}
				foundAfterEndDate = true;
			}
			snapshots.add(readEnrollmentSnapshot(file));
		}
		if (snapshots.size() < 2) {
			throw new IOException("Insufficient snapshots in " + semesterFolder.getFileName());
		}
		return snapshots;
	}
	
}
